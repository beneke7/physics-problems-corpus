(c) Starting with $\frac{d V}{d t}=-\frac{k}{\left(T-T_{F}\right.}$

$$
\begin{aligned}
\text { Shorthand } \frac{d V}{d t} & =\dot{V} ; \frac{d T}{d t}=T \\
\frac{d^{2} V}{d t^{2}} & =\dot{V}
\end{aligned}
$$

$$
\begin{aligned}
\Rightarrow \frac{d^{2} V}{d t^{2}} & =\frac{k}{\left(T-T_{F}\right)^{2}} \frac{d T}{d t} \\
& =\frac{1}{\dot{k}}\left(\frac{d V}{d t}\right)^{2} \frac{d T}{d t} \text { using } \frac{1}{T^{2}-T_{F}}=-\frac{1}{k} \frac{d V}{d t} \\
& =-\frac{h A}{\rho c k} \frac{1}{V}\left(\frac{d V}{d t}\right)^{2}\left(T-T_{R}\right) \text { using } \frac{d T}{d t}=-\frac{h A}{\rho c} \frac{\left(T-T_{R}\right)}{V} \\
& =-\frac{h A}{\rho c k} \frac{\dot{V}^{2}}{V}\left(-\frac{k}{\dot{V}}+T_{F}-T_{R}\right) \text { using } \dot{V}=\frac{-k}{T-T_{F}} \\
\ddot{V} & =+\frac{h A}{\rho c} \frac{\dot{V}}{V}-\frac{h A}{\rho c k}\left(T_{F}-T_{R}\right) \frac{\dot{V}^{2}}{V}
\end{aligned}
$$

Using suggested relationship $\ddot{V}=\dot{V} \frac{d \dot{V}}{d V}$ \& let $m=\frac{h A}{\rho c}$
gives $\dot{V} \frac{d V}{d V}=\frac{m V}{V}-\frac{m}{k}\left(T_{F}-T_{R}\right) \frac{V^{k}}{V}$

$$
\begin{aligned}
& \Rightarrow \frac{d \dot{V}}{d V}=\frac{m}{V}\left(1-\frac{T_{F}-T_{R}}{k} \dot{V}\right) \\
& \Rightarrow \int_{P} \frac{d \dot{V}}{1-\alpha \dot{V}}=\int_{V_{0}}^{m} \frac{d V}{V} \quad\left(\ln \ln \alpha=\frac{T_{F}-T_{R}}{k}\right) \\
& \Rightarrow\left[-\frac{1}{\alpha} \ln (1-\alpha \dot{V})\right]_{\dot{V}_{0}}^{V}=m \ln \left(\frac{V}{V_{0}}\right) \\
& \Rightarrow-\frac{1}{\alpha} \ln \left(\frac{1-\alpha \dot{V}}{1-\alpha \dot{V}_{0}}\right)=m \ln \left(\frac{V}{V_{0}}\right) \\
& \Rightarrow \ln \left(\frac{V}{V_{0}}\right)=\ln \left[\left(\left(1+\frac{\left(T_{F}-T_{R}\right)}{k} \cdot \frac{k}{T_{-}-T_{F}}\right)\left(1+\frac{\left(T_{F}-T_{R}\right)}{k} \cdot \frac{k}{T_{0}-T_{F}}\right)^{-\frac{1}{\alpha m}}\right]\right. \\
& \quad=\ln \left[\left(\frac{T-T_{F}^{\prime}+T_{F}-T_{R}}{T-T_{F}} \cdot \frac{T_{0}-T_{F}}{T_{0}-T_{F}+T_{F}-T_{R}}\right)^{-\frac{1}{\alpha m}}\right] \\
& \Rightarrow V=V_{0}\left(\frac{\left(T_{0}-T_{R}\right)\left(T-T_{F}\right)}{\left(T_{0}-T_{F}\right)\left(T-T_{R}\right)}\right)^{\frac{k}{m\left(T_{F}-T_{R}\right)}} \quad\left(m=\frac{h A}{\rho C}\right)
\end{aligned}
$$
